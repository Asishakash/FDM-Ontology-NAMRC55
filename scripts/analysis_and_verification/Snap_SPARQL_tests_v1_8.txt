TEST 20  CQ9  [Req01]  Which recommended setting prints fastest (then least filament)?
----------------------------------------------------------------------------------------------------
PREFIX f: <http://www.wcu.edu/ontology/fdm-quality#>
SELECT ?ps ?printTime_s ?filament_g ?worstMargin ?confidence WHERE {
 { SELECT ?ps ?printTime_s ?filament_g (MIN(?m) AS ?worstMargin) WHERE {
  ?ps f:recommendsSettingFor f:Req01 ; f:estimatedPrintTimeValue ?printTime_s ; f:estimatedFilamentMassValue ?filament_g .

  ?e f:evaluatesSetting ?ps ; f:forRequirement f:Req01 ; f:coversAssessment ?a .
  ?a f:assessesResult ?r .
  ?sp f:hasResult ?r .
  { f:Req01 f:requiredMaxLengthDeviation ?lim . ?r f:absLengthDeviationValue ?v . BIND("length" AS ?char) }
  UNION { f:Req01 f:requiredMaxWidthDeviation ?lim . ?r f:absWidthDeviationValue ?v . BIND("width" AS ?char) }
  UNION { f:Req01 f:requiredMaxHeightDeviation ?lim . ?r f:absHeightDeviationValue ?v . BIND("height" AS ?char) }
  UNION { f:Req01 f:requiredMaxRaValue ?lim . ?r f:surfaceRoughnessRaValue ?v . BIND("top Ra" AS ?char) }
  BIND((?lim - ?v) / ?lim AS ?m)
 } GROUP BY ?ps ?printTime_s ?filament_g }
 BIND(IF(?worstMargin < 0.1, "marginal", "clear") AS ?confidence)
} ORDER BY ?printTime_s ?filament_g DESC(?worstMargin)
----------------------------------------------------------------------------------------------------
EXPECTED: 5 rows: PS07 212 2.31 0.314 clear; PS06 288 2.21 0.52 clear; PS04 293 2.33 0.29 clear; PS01 525 2.25 0.288 clear; PS03 528 2.46 0.146 clear   (answer = first row: PS07)
